Usage: {{< gallery folder="photos" >}} {{< gallery folder="photos" thumb="320x" >}} */}} {{ $folder := .Get "folder" | default "gallery" }} {{ $thumb := .Get "thumb" }} {{ $images := $.Page.Resources.Match (printf "%s/*" $folder) }}
